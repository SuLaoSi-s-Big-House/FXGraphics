CMAKE_MINIMUM_REQUIRED(VERSION 3.12.0 FATAL_ERROR)

IF(NOT DEFINED FXG_ROOT_DIR)
    MESSAGE(FATAL_ERROR "The project should be built from the top-level directory.")
ENDIF()

SET(PROJECT_NAME shader)
PROJECT(${PROJECT_NAME})

FILE(GLOB SHADER_FILES ./*.h ./*.vert ./*.frag ./*.geom)

ADD_CUSTOM_TARGET(${PROJECT_NAME} SOURCES
    ${SHADER_FILES}
)

SET(TARGET_DIR ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>/shader)

IF(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
    ADD_CUSTOM_COMMAND(TARGET ${PROJECT_NAME}
        COMMAND ${CMAKE_COMMAND} -E copy ${SHADER_FILES} ${TARGET_DIR}
        COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/expandw.exe ${TARGET_DIR}
        COMMAND expandw.exe
        COMMAND del *.exe
        COMMAND del *.h
        WORKING_DIRECTORY ${TARGET_DIR}
    )
ELSE()
    MESSAGE(WARNING "New platform not adapted!")
ENDIF()

SET_TARGET_PROPERTIES(${PROJECT_NAME} PROPERTIES FOLDER "resources")
