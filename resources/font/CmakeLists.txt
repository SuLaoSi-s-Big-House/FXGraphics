CMAKE_MINIMUM_REQUIRED(VERSION 3.12.0 FATAL_ERROR)

IF(NOT DEFINED FXG_ROOT_DIR)
    MESSAGE(FATAL_ERROR "The project should be built from the top-level directory.")
ENDIF()

SET(PROJECT_NAME font)
PROJECT(${PROJECT_NAME})

FILE(GLOB FONT_FILES ./*.ttf)

ADD_CUSTOM_TARGET(${PROJECT_NAME} SOURCES
    ${FONT_FILES}
)

SET(TARGET_DIR ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>/font)

ADD_CUSTOM_COMMAND(TARGET ${PROJECT_NAME}
    COMMAND ${CMAKE_COMMAND} -E copy ${FONT_FILES} ${TARGET_DIR}
)

SET_TARGET_PROPERTIES(${PROJECT_NAME} PROPERTIES FOLDER "resources")
