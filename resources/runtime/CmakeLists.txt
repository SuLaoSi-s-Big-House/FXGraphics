CMAKE_MINIMUM_REQUIRED(VERSION 3.12.0 FATAL_ERROR)

IF(NOT DEFINED FXG_ROOT_DIR)
    MESSAGE(FATAL_ERROR "The project should be built from the top-level directory.")
ENDIF()

SET(PROJECT_NAME runtime)
PROJECT(${PROJECT_NAME})

ADD_CUSTOM_TARGET(${PROJECT_NAME})

SET(TARGET_DIR ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>)

IF(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")

    IF(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    ELSEIF(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
        SET(RUNTIME_LIB
            ${CMAKE_CURRENT_SOURCE_DIR}/libgcc_s_seh-1.dll
            ${CMAKE_CURRENT_SOURCE_DIR}/libstdc++-6.dll
            ${CMAKE_CURRENT_SOURCE_DIR}/libwinpthread-1.dll
        )

        ADD_CUSTOM_COMMAND(TARGET ${PROJECT_NAME} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different ${RUNTIME_LIB} ${TARGET_DIR}
        )
    ELSE()
        MESSAGE(WARNING "New IDE not adapted!")
    ENDIF()

ELSE()
    MESSAGE(WARNING "New platform not adapted!")
ENDIF()

SET_TARGET_PROPERTIES(${PROJECT_NAME} PROPERTIES FOLDER "resources")
